const SPREADSHEET_ID = '1-ksbEhSa-xIDIyqT3BzEvGB8WbrFDfZQqYhrOeoJhXw'; function doGet(e) { // GET 요청시 현재 설정값과 데이터(단어 빈도수)를 JSON으로 반환 const action = e.parameter.action; if (action === 'getState') { return getState(); } return ContentService.createTextOutput(JSON.stringify({error: 'Invalid GET action'})).setMimeType(ContentService.MimeType.JSON); } function doPost(e) { // POST 요청 처리 (단어 제출, 데이터 초기화, 질문 수정) // CORS Preflight를 피하기 위해 text/plain으로 요청을 받습니다. let params; try { params = JSON.parse(e.postData.contents); } catch (error) { return ContentService.createTextOutput(JSON.stringify({error: 'Invalid JSON payload'})).setMimeType(ContentService.MimeType.JSON); } const action = params.action; if (action === 'submitWords') { return submitWords(params.words); } else if (action === 'resetData') { return resetData(params.password); } else if (action === 'updateQuestion') { return updateQuestion(params.password, params.question); } return ContentService.createTextOutput(JSON.stringify({error: 'Invalid POST action'})).setMimeType(ContentService.MimeType.JSON); } function getState() { const ss = SpreadsheetApp.openById(SPREADSHEET_ID); const settingsSheet = ss.getSheetByName('설정'); const dataSheet = ss.getSheetByName('데이터'); // 설정 시트 데이터 가져오기 (A2:E2) const settings = settingsSheet.getRange('A2:E2').getValues()[0]; const password = settings[0]; const title = settings[1]; const target = settings[2]; const numInputs = settings[3]; const question = settings[4]; // 데이터 시트 단어 가져오기 (1행은 헤더이므로 제외, A2부터) const lastRow = dataSheet.getLastRow(); let words = []; if (lastRow > 1) { const dataRange = dataSheet.getRange(2, 1, lastRow - 1, 1); // 빈 셀 제외하고 1차원 배열로 평탄화 words = dataRange.getValues().flat().filter(String); } // 서버에서 단어 빈도수를 계산하여 전송량 최소화 (빠른 동기화를 위함) const freq = {}; words.forEach(w => { const trimmed = w.toString().trim(); if(trimmed.length > 0) { freq[trimmed] = (freq[trimmed] || 0) + 1; } }); const result = { title: title, target: target, numInputs: parseInt(numInputs, 10) || 1, question: question, words: freq }; // 캐시 방지 및 JSON 응답 반환 const output = ContentService.createTextOutput(JSON.stringify(result)); output.setMimeType(ContentService.MimeType.JSON); return output; } function submitWords(words) { const ss = SpreadsheetApp.openById(SPREADSHEET_ID); const dataSheet = ss.getSheetByName('데이터'); const rows = words.filter(String).map(w => [w.toString().trim()]); if (rows.length > 0) { // 마지막 행 다음부터 제출된 단어들 추가 dataSheet.getRange(dataSheet.getLastRow() + 1, 1, rows.length, 1).setValues(rows); } return ContentService.createTextOutput(JSON.stringify({success: true})).setMimeType(ContentService.MimeType.JSON); } function resetData(password) { const ss = SpreadsheetApp.openById(SPREADSHEET_ID); const settingsSheet = ss.getSheetByName('설정'); const realPassword = settingsSheet.getRange('A2').getValue(); if (password.toString() === realPassword.toString()) { const dataSheet = ss.getSheetByName('데이터'); const lastRow = dataSheet.getLastRow(); // 1행(헤더)을 제외하고 A2부터 일괄 삭제 if (lastRow > 1) { dataSheet.getRange(2, 1, lastRow - 1, 1).clearContent(); } return ContentService.createTextOutput(JSON.stringify({success: true})).setMimeType(ContentService.MimeType.JSON); } else { return ContentService.createTextOutput(JSON.stringify({success: false, error: '인증번호가 일치하지 않습니다.'})).setMimeType(ContentService.MimeType.JSON); } } function updateQuestion(password, question) { const ss = SpreadsheetApp.openById(SPREADSHEET_ID); const settingsSheet = ss.getSheetByName('설정'); const realPassword = settingsSheet.getRange('A2').getValue(); if (password.toString() === realPassword.toString()) { // E2 셀에 새로운 질문 반영 settingsSheet.getRange('E2').setValue(question); return ContentService.createTextOutput(JSON.stringify({success: true})).setMimeType(ContentService.MimeType.JSON); } else { return ContentService.createTextOutput(JSON.stringify({success: false, error: '인증번호가 일치하지 않습니다.'})).setMimeType(ContentService.MimeType.JSON); } }
